# 1) Create Docker network (run once; ignore if already exists)
docker network create pg-network

# 2) Start PostgreSQL container (keep this terminal running)
docker run -it --rm `
  -e POSTGRES_USER=root `
  -e POSTGRES_PASSWORD=root `
  -e POSTGRES_DB=ny_taxi `
  -e POSTGRES_HOST_AUTH_METHOD=trust `
  -v ny_taxi:/var/lib/postgresql/data `
  -p 5433:5432 `
  --network=pg-network `
  --name pgdatabase `
  postgres:17

# 3) (Optional) Start pgAdmin in another terminal
docker run -it `
  -e PGADMIN_DEFAULT_EMAIL=admin@admin.com `
  -e PGADMIN_DEFAULT_PASSWORD=root `
  -v pgadmin_data:/var/lib/pgadmin `
  -p 8085:80 `
  --network=pg-network `
  --name pgadmin `
  dpage/pgadmin4

# 4) Run ingest container (another terminal)
docker run -it --rm `
  --network=pg-network `
  taxi_ingest:v001 `
  --year=2021 `
  --month=1 `
  --pg-user=root `
  --pg-password=root `
  --pg-host=pgdatabase `
  --pg-port=5432 `
  --pg-db=ny_taxi `
  --chunksize=100000

# 5) Connect with pgcli from host machine (another terminal)
uv run pgcli -h localhost -p 5433 -u root -d ny_taxi
